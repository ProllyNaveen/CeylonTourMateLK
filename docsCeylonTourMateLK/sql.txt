CREATE DATABASE IF NOT EXISTS ceylontourmatelk;
USE ceylontourmatelk;

-- ===========================
-- USERS
-- ===========================
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    firstname VARCHAR(50),
    lastname VARCHAR(50),
    username VARCHAR(50) UNIQUE,
    email VARCHAR(100) UNIQUE,
    password VARCHAR(255),
    phone VARCHAR(15),
    profile_pic VARCHAR(255) DEFAULT 'default.png',
    role ENUM('tourist', 'tour_guide', 'shop_owner', 'hotel_owner', 'experience_provider', 'admin'),
    is_suspended TINYINT DEFAULT 0,
    is_approved TINYINT DEFAULT 0,
    security_question VARCHAR(255),
    security_answer VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ===========================
-- LISTINGS (parent)
-- ===========================
CREATE TABLE listings (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    title VARCHAR(255),
    description TEXT,
    location VARCHAR(255),
    city VARCHAR(100),
    price DECIMAL(10,2),
    status ENUM('pending', 'approved', 'rejected') DEFAULT 'pending',
    type ENUM('tour_guide', 'shop', 'hotel', 'experience'),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id)
);
-- ===========================
-- LISTINGS (disjoint subtypes)
-- ===========================
CREATE TABLE tourguide_listings (
    id INT AUTO_INCREMENT PRIMARY KEY,
    listing_id INT,
    languages VARCHAR(255),
    specialization VARCHAR(255),
    experience_years INT,
    availability VARCHAR(255),
    FOREIGN KEY (listing_id) REFERENCES listings(id)
);

CREATE TABLE shop_listings (
    id INT AUTO_INCREMENT PRIMARY KEY,
    listing_id INT,
    shop_type VARCHAR(100),
    open_time TIME,
    close_time TIME,
    FOREIGN KEY (listing_id) REFERENCES listings(id)
);

CREATE TABLE hotel_listings (
    id INT AUTO_INCREMENT PRIMARY KEY,
    listing_id INT,
    star_rating INT,
    amenities TEXT,
    total_rooms INT,
    check_in_time TIME,
    check_out_time TIME,
    FOREIGN KEY (listing_id) REFERENCES listings(id)
);

CREATE TABLE experience_listings (
    id INT AUTO_INCREMENT PRIMARY KEY,
    listing_id INT,
    experience_type VARCHAR(100),
    duration VARCHAR(100),
    max_group_size INT,
    availability VARCHAR(255),
    FOREIGN KEY (listing_id) REFERENCES listings(id)
);

-- ===========================
-- LISTING IMAGES
-- ===========================
CREATE TABLE listing_images (
    id INT AUTO_INCREMENT PRIMARY KEY,
    listing_id INT,
    image_path VARCHAR(255),
    FOREIGN KEY (listing_id) REFERENCES listings(id)
);

-- ===========================
-- ROOMS (belongs to hotel_listings)
-- ===========================
CREATE TABLE rooms (
    id INT AUTO_INCREMENT PRIMARY KEY,
    hotel_listing_id INT,
    description TEXT,
    room_type ENUM('single', 'double', 'suite', 'family'),
    price_per_night DECIMAL(10,2),
    capacity INT,
    is_available TINYINT DEFAULT 1,
    FOREIGN KEY (hotel_listing_id) REFERENCES hotel_listings(id)
);

-- ===========================
-- BOOKINGS (single unified table)
-- room_id only used for hotel bookings, NULL otherwise
-- ===========================
CREATE TABLE bookings (
    id INT AUTO_INCREMENT PRIMARY KEY,
    tourist_id INT,
    listing_id INT,
    room_id INT NULL,
    status ENUM('pending', 'confirmed', 'cancelled', 'completed') DEFAULT 'pending',
    start_datetime DATETIME,
    end_datetime DATETIME,
    guest_count INT,
    total_price DECIMAL(10,2),
    commission_amount DECIMAL(10,2),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (tourist_id) REFERENCES users(id),
    FOREIGN KEY (listing_id) REFERENCES listings(id),
    FOREIGN KEY (room_id) REFERENCES rooms(id)
);

-- ===========================
-- PAYMENTS (single unified table)
-- shop_listing_id used only when type = 'listing_fee'
-- booking_id used only when type = 'booking'
-- ===========================
CREATE TABLE payments (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    shop_listing_id INT NULL,
    booking_id INT NULL,
    type ENUM('booking', 'listing_fee'),
    amount DECIMAL(10,2),
    payment_status ENUM('pending', 'completed', 'failed') DEFAULT 'pending',
    fee_month INT NULL,
    fee_year INT NULL,
    payment_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id),
    FOREIGN KEY (shop_listing_id) REFERENCES shop_listings(id),
    FOREIGN KEY (booking_id) REFERENCES bookings(id)
);

-- ===========================
-- REVIEWS
-- ===========================
CREATE TABLE reviews (
    id INT AUTO_INCREMENT PRIMARY KEY,
    tourist_id INT,
    listing_id INT,
    booking_id INT,
    rating INT CHECK (rating BETWEEN 1 AND 5),
    comment TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (tourist_id) REFERENCES users(id),
    FOREIGN KEY (listing_id) REFERENCES listings(id),
    FOREIGN KEY (booking_id) REFERENCES bookings(id)
);

-- ===========================
-- MESSAGES
-- ===========================
CREATE TABLE messages (
    id INT AUTO_INCREMENT PRIMARY KEY,
    sender_id INT,
    receiver_id INT,
    listing_id INT,
    message TEXT,
    is_read TINYINT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (sender_id) REFERENCES users(id),
    FOREIGN KEY (receiver_id) REFERENCES users(id),
    FOREIGN KEY (listing_id) REFERENCES listings(id)
);

-- ===========================
-- NOTIFICATIONS
-- ===========================
CREATE TABLE notifications (
    id INT AUTO_INCREMENT PRIMARY KEY,
    sender_id INT,
    receiver_id INT,
    message TEXT,
    is_read TINYINT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (sender_id) REFERENCES users(id),
    FOREIGN KEY (receiver_id) REFERENCES users(id)
);

-- ===========================
-- COMMISSION (rate history)
-- ===========================
CREATE TABLE commission (
    id INT AUTO_INCREMENT PRIMARY KEY,
    rate DECIMAL(5,2),
    updated_by INT,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (updated_by) REFERENCES users(id)
);

-- ===========================
-- TOURGUIDE LICENSES
-- ===========================
CREATE TABLE tourguide_licenses (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    license_number VARCHAR(100),
    license_image VARCHAR(255),
    FOREIGN KEY (user_id) REFERENCES users(id)
);
